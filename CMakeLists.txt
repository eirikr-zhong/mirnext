cmake_minimum_required(VERSION 3.14)
project(MIRNext LANGUAGES CXX)

include(CTest)
include(GNUInstallDirs)

option(BUILD_SHARED_LIBS "Build shared libraries instead of static libraries" OFF)
option(MIRNEXT_BUILD_UTILITIES "Build MIR format and execution commands" ON)
option(MIRNEXT_BUILD_BENCHMARKS "Build C++ microbenchmarks" OFF)
option(MIRNEXT_BUILD_EXAMPLES "Build C++ API examples" ON)
option(MIRNEXT_BUILD_SCRIPT_TESTS "Register Python integration checks" ON)
option(MIRNEXT_ENABLE_ASAN "Build with AddressSanitizer" OFF)
option(MIRNEXT_ENABLE_NATIVE_EVIDENCE "Build source-stamped platform acceptance tests" OFF)
set(MIRNEXT_EMBEDDED_HEADER "" CACHE FILEPATH "Optional b2cpp output for mir-embedded-run")
include(cmake/MIRNextBuild.cmake)

set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Keep executable and library paths stable when building from subdirectories.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")

if(BUILD_TESTING AND MIRNEXT_ENABLE_NATIVE_EVIDENCE AND NOT MIRNEXT_BUILD_SCRIPT_TESTS)
  message(FATAL_ERROR "MIRNEXT_ENABLE_NATIVE_EVIDENCE requires MIRNEXT_BUILD_SCRIPT_TESTS")
endif()
if(BUILD_TESTING AND MIRNEXT_BUILD_SCRIPT_TESTS)
  find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
endif()

add_subdirectory(lib)
if(MIRNEXT_BUILD_UTILITIES)
  add_subdirectory(tools)
endif()
if(MIRNEXT_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(MIRNEXT_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
