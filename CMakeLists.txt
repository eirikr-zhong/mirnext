function(mir_target_common_options target)
  if(NOT WIN32)
    target_compile_options(${target} PRIVATE
      $<$<CONFIG:RELEASE>:-O3>
      $<$<CONFIG:RelWithDebInfo>:-O3>
      -fsigned-char
      -fPIC
    )
  endif()
endfunction()

function(mir_target_c_options target)
  if(NOT WIN32)
    target_compile_options(${target} PRIVATE -std=gnu11 -Wno-abi ${MIR_C_WORKAROUND_FLAGS})
  endif()
endfunction()

function(mir_strip_msvc_exception_flags variable)
  if(DEFINED ${variable})
    string(REGEX REPLACE "(^| )[/-]EH[a-zA-Z0-9+-]*" "" ${variable} "${${variable}}")
    string(STRIP "${${variable}}" ${variable})
    set(${variable} "${${variable}}" PARENT_SCOPE)
  endif()
endfunction()

function(mir_target_cxx_options target)
  if(MSVC)
    set(rtti_option /GR-)
    if(ARGV1 STREQUAL "WITH_RTTI")
      set(rtti_option /GR)
    endif()
    target_compile_definitions(${target} PRIVATE _HAS_EXCEPTIONS=0)
    target_compile_options(${target} PRIVATE
      $<$<COMPILE_LANGUAGE:CXX>:/EHs->
      $<$<COMPILE_LANGUAGE:CXX>:/EHc->
      $<$<COMPILE_LANGUAGE:CXX>:${rtti_option}>)
  else()
    set(rtti_option -fno-rtti)
    if(ARGV1 STREQUAL "WITH_RTTI")
      set(rtti_option -frtti)
    endif()
    target_compile_options(${target} PRIVATE
      $<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>
      $<$<COMPILE_LANGUAGE:CXX>:${rtti_option}>)
  endif()
endfunction()

# The replacement architecture is built independently of the pre-rewrite runtime.
# Its include path intentionally contains no legacy engine or bridge headers.
function(mirnext_add_layer target exported_name)
  foreach(source IN LISTS ARGN)
    if(NOT source MATCHES "^lib/[^;]+[.]cpp$")
      message(FATAL_ERROR "Product layer ${target} requires owning C++ sources under lib: ${source}")
    endif()
  endforeach()
  # Compile a layer once. Both linkage variants own exactly these objects, and
  # their public dependency graphs stay entirely static or entirely shared.
  add_library(${target}_objects OBJECT ${ARGN})
  target_include_directories(${target}_objects PRIVATE ${PROJECT_SOURCE_DIR}/include)
  target_compile_features(${target}_objects PRIVATE cxx_std_20)
  set_target_properties(${target}_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
  mir_target_common_options(${target}_objects)
  mir_target_cxx_options(${target}_objects)
  add_library(${target} STATIC $<TARGET_OBJECTS:${target}_objects>)
  add_library(${target}_shared SHARED $<TARGET_OBJECTS:${target}_objects>)
  add_library(MIRNext::${exported_name} ALIAS ${target})
  add_library(MIRNext::${exported_name}Shared ALIAS ${target}_shared)
  foreach(variant IN ITEMS ${target} ${target}_shared)
    target_include_directories(${variant} PUBLIC
      $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
    target_compile_features(${variant} PUBLIC cxx_std_20)
    set_property(TARGET ${variant} APPEND PROPERTY COMPATIBLE_INTERFACE_STRING MIRNEXT_LINKAGE)
  endforeach()
  set_target_properties(${target} PROPERTIES INTERFACE_MIRNEXT_LINKAGE Static)
  set_target_properties(${target}_shared PROPERTIES INTERFACE_MIRNEXT_LINKAGE Shared)
  set_target_properties(${target} PROPERTIES EXPORT_NAME ${exported_name})
  set_target_properties(${target}_shared PROPERTIES
    EXPORT_NAME ${exported_name}Shared WINDOWS_EXPORT_ALL_SYMBOLS TRUE)
  # Each installed DSO finds its own adjacent dependencies, including transitive
  # dependencies on ELF systems where an executable's RUNPATH is not inherited.
  if(APPLE)
    set_target_properties(${target}_shared PROPERTIES INSTALL_RPATH "@loader_path")
  elseif(UNIX)
    set_target_properties(${target}_shared PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
endfunction()

function(mirnext_layer_dependencies target)
  foreach(dependency IN LISTS ARGN)
    target_link_libraries(${target} PUBLIC ${dependency})
    target_link_libraries(${target}_shared PUBLIC ${dependency}_shared)
  endforeach()
endfunction()
