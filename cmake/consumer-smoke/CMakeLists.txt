cmake_minimum_required(VERSION 3.14)
project(MIRNextConsumerSmoke LANGUAGES CXX)
# Linking MIRNext must raise this to C++20 through its public requirements.
set(CMAKE_CXX_STANDARD 11)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(MIRNEXT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MIRNEXT_BUILD_C_REFERENCE OFF CACHE BOOL "" FORCE)
set(MIRNEXT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
get_filename_component(MIRNEXT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${MIRNEXT_SOURCE_DIR}" mirnext EXCLUDE_FROM_ALL)
add_executable(consumer_static "${MIRNEXT_SOURCE_DIR}/examples/native-variables.cpp")
target_link_libraries(consumer_static PRIVATE MIRNext::Serialization)
add_executable(consumer_shared "${MIRNEXT_SOURCE_DIR}/examples/native-calls.cpp")
target_link_libraries(consumer_shared PRIVATE MIRNext::SerializationShared)
foreach(linkage static shared)
  add_executable(consumer_session_${linkage} "${MIRNEXT_SOURCE_DIR}/examples/native-session.cpp")
  add_executable(consumer_mixed_${linkage} "${MIRNEXT_SOURCE_DIR}/examples/mixed-engines.cpp")
endforeach()
target_link_libraries(consumer_session_static PRIVATE MIRNext::Execution)
target_link_libraries(consumer_session_shared PRIVATE MIRNext::ExecutionShared)
target_link_libraries(consumer_mixed_static PRIVATE MIRNext::Execution)
target_link_libraries(consumer_mixed_shared PRIVATE MIRNext::ExecutionShared)
enable_testing()
add_test(NAME consumer-static COMMAND consumer_static)
add_test(NAME consumer-shared COMMAND consumer_shared)
add_test(NAME consumer-session-static COMMAND consumer_session_static)
add_test(NAME consumer-session-shared COMMAND consumer_session_shared)
add_test(NAME consumer-mixed-static COMMAND consumer_mixed_static)
add_test(NAME consumer-mixed-shared COMMAND consumer_mixed_shared)
if(WIN32)
  string(REPLACE ";" "\\;" consumer_path "$ENV{PATH}")
  set_tests_properties(consumer-shared consumer-session-shared consumer-mixed-shared PROPERTIES
    ENVIRONMENT "PATH=$<TARGET_FILE_DIR:MIRNext::ExecutionShared>\\;${consumer_path}")
endif()
