foreach(name IN ITEMS m2b b2m b2cpp)
  add_executable(${name} Format.cpp)
  target_link_libraries(${name} PRIVATE MIRNext::MIRNext)
  mir_target_cxx_options(${name})
endforeach()
target_compile_definitions(m2b PRIVATE MIRNEXT_FORMAT_MODE=0)
target_compile_definitions(b2m PRIVATE MIRNEXT_FORMAT_MODE=1)
target_compile_definitions(b2cpp PRIVATE MIRNEXT_FORMAT_MODE=2)
add_executable(mir-bin-run BinaryRun.cpp Run.cpp)
target_link_libraries(mir-bin-run PRIVATE MIRNext::MIRNext ${CMAKE_DL_LIBS})
mir_target_cxx_options(mir-bin-run)
if(BUILD_SHARED_LIBS AND UNIX)
  file(RELATIVE_PATH tool_library_path "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
  if(APPLE)
    set(tool_install_rpath "@loader_path/${tool_library_path}")
  else()
    set(tool_install_rpath "$ORIGIN/${tool_library_path}")
  endif()
  set_target_properties(m2b b2m b2cpp mir-bin-run PROPERTIES INSTALL_RPATH "${tool_install_rpath}")
endif()
install(TARGETS m2b b2m b2cpp mir-bin-run RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(MIRNEXT_EMBEDDED_HEADER)
  get_filename_component(embedded_header "${MIRNEXT_EMBEDDED_HEADER}" ABSOLUTE)
  add_executable(mir-embedded-run EmbeddedRun.cpp Run.cpp)
  target_compile_definitions(mir-embedded-run PRIVATE MIRNEXT_EMBEDDED_HEADER="${embedded_header}")
  target_link_libraries(mir-embedded-run PRIVATE MIRNext::MIRNext ${CMAKE_DL_LIBS})
  mir_target_cxx_options(mir-embedded-run)
endif()
