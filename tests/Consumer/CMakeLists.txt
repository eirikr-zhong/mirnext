cmake_minimum_required(VERSION 3.14)
project(MIRNextNativeConsumer LANGUAGES CXX)
get_property(consumer_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
find_package(MIRNext CONFIG REQUIRED)
get_property(package_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
if(NOT package_languages STREQUAL consumer_languages)
  message(FATAL_ERROR "MIRNext must not enable additional languages in its consumer")
endif()
enable_testing()
get_target_property(expected_type MIRNext::MIRNext TYPE)
if(NOT expected_type MATCHES "^(STATIC|SHARED)_LIBRARY$")
  message(FATAL_ERROR "Unexpected installed library type: ${expected_type}")
endif()
if(DEFINED BUILD_SHARED_LIBS)
  if(BUILD_SHARED_LIBS)
    set(requested_type SHARED_LIBRARY)
  else()
    set(requested_type STATIC_LIBRARY)
  endif()
  if(NOT expected_type STREQUAL requested_type)
    message(FATAL_ERROR "Installed package has type ${expected_type}, requested ${requested_type}")
  endif()
endif()
add_executable(consumer main.cpp)
add_executable(builder-consumer Builder.cpp)
foreach(target consumer builder-consumer)
  target_link_libraries(${target} PRIVATE MIRNext::MIRNext)
  if(MSVC)
    target_compile_options(${target} PRIVATE /EHs-c- /GR-)
    target_compile_definitions(${target} PRIVATE _HAS_EXCEPTIONS=0)
  else()
    target_compile_options(${target} PRIVATE -fno-exceptions -fno-rtti)
  endif()
endforeach()
add_test(NAME native-consumer COMMAND consumer)
add_test(NAME builder-consumer COMMAND builder-consumer)
if(WIN32 AND expected_type STREQUAL "SHARED_LIBRARY")
  string(REPLACE ";" "\\;" consumer_path "$ENV{PATH}")
  set_tests_properties(native-consumer builder-consumer PROPERTIES
    ENVIRONMENT "PATH=$<TARGET_FILE_DIR:MIRNext::MIRNext>\\;${consumer_path}")
endif()
