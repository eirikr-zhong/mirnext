cmake_minimum_required(VERSION 3.14)
project(MIRNextNativeConsumer LANGUAGES CXX)
find_package(MIRNext CONFIG REQUIRED)
get_property(consumer_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
if("C" IN_LIST consumer_languages)
  message(FATAL_ERROR "The installed C++ package enabled the C language")
endif()
enable_testing()
# Inspect every installed layer, including dependencies reached only through the
# Execution target. A shared session must not embed another layer's static copy.
foreach(layer IN ITEMS IR Analysis Transforms Target CodeGen Serialization Execution)
  foreach(suffix IN ITEMS "" Shared)
    set(layer_target MIRNext::${layer}${suffix})
    if(NOT TARGET ${layer_target})
      message(FATAL_ERROR "Missing installed layer: ${layer_target}")
    endif()
    get_target_property(kind ${layer_target} TYPE)
    if(suffix STREQUAL "Shared")
      set(expected_type SHARED_LIBRARY)
    else()
      set(expected_type STATIC_LIBRARY)
    endif()
    if(NOT kind STREQUAL expected_type)
      message(FATAL_ERROR "${layer_target} has type ${kind}, expected ${expected_type}")
    endif()
    get_target_property(dependencies ${layer_target} INTERFACE_LINK_LIBRARIES)
    foreach(dependency IN LISTS dependencies)
      if(dependency MATCHES "^MIRNext::")
        if(suffix STREQUAL "Shared" AND NOT dependency MATCHES "Shared$")
          message(FATAL_ERROR "${layer_target} embeds a static layer: ${dependency}")
        elseif(suffix STREQUAL "" AND dependency MATCHES "Shared$")
          message(FATAL_ERROR "${layer_target} depends on a shared layer: ${dependency}")
        endif()
      endif()
    endforeach()
  endforeach()
endforeach()
foreach(linkage IN ITEMS static shared)
  add_executable(consumer_${linkage} main.cpp)
  target_compile_features(consumer_${linkage} PRIVATE cxx_std_20)
  if(linkage STREQUAL "static")
    target_link_libraries(consumer_${linkage} PRIVATE MIRNext::Execution)
  else()
    target_link_libraries(consumer_${linkage} PRIVATE MIRNext::ExecutionShared)
  endif()
  if(MSVC)
    target_compile_options(consumer_${linkage} PRIVATE /EHs- /EHc- /GR-)
    target_compile_definitions(consumer_${linkage} PRIVATE _HAS_EXCEPTIONS=0)
  else()
    target_compile_options(consumer_${linkage} PRIVATE -fno-exceptions -fno-rtti)
  endif()
  add_test(NAME native-consumer-${linkage} COMMAND consumer_${linkage})
endforeach()
if(WIN32)
  # Installed DLLs live beside each other in bin, outside the consumer build.
  string(REPLACE ";" "\\;" consumer_path "$ENV{PATH}")
  set_tests_properties(native-consumer-shared PROPERTIES
    ENVIRONMENT "PATH=$<TARGET_FILE_DIR:MIRNext::ExecutionShared>\\;${consumer_path}")
endif()
add_test(NAME native-consumer-reject-mixed-linkage
  COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/Incompatible
    -DBINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}/incompatible -DMIRNext_DIR=${MIRNext_DIR}
    -DCXX_COMPILER=${CMAKE_CXX_COMPILER} -DGENERATOR=${CMAKE_GENERATOR}
    -DGENERATOR_PLATFORM=${CMAKE_GENERATOR_PLATFORM} -DGENERATOR_TOOLSET=${CMAKE_GENERATOR_TOOLSET}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/CheckLinkage.cmake)
