cmake_minimum_required(VERSION 3.14)
project(MIRNextConsumerSmoke LANGUAGES CXX)
# Linking MIRNext must raise this to C++20 through its public requirements.
set(CMAKE_CXX_STANDARD 11)
set(BUILD_TESTING OFF)
set(MIRNEXT_BUILD_EXAMPLES OFF)
get_filename_component(MIRNEXT_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${MIRNEXT_SOURCE_DIR}" mirnext EXCLUDE_FROM_ALL)
add_executable(consumer_variables "${MIRNEXT_SOURCE_DIR}/examples/native-variables.cpp")
target_link_libraries(consumer_variables PRIVATE MIRNext::MIRNext)
add_executable(consumer_calls "${MIRNEXT_SOURCE_DIR}/examples/native-calls.cpp")
target_link_libraries(consumer_calls PRIVATE MIRNext::MIRNext)
add_executable(consumer_session "${MIRNEXT_SOURCE_DIR}/examples/native-session.cpp")
target_link_libraries(consumer_session PRIVATE MIRNext::MIRNext)
add_executable(consumer_mixed "${MIRNEXT_SOURCE_DIR}/examples/mixed-engines.cpp")
target_link_libraries(consumer_mixed PRIVATE MIRNext::MIRNext)
enable_testing()
add_test(NAME consumer-variables COMMAND consumer_variables)
add_test(NAME consumer-calls COMMAND consumer_calls)
add_test(NAME consumer-session COMMAND consumer_session)
add_test(NAME consumer-mixed COMMAND consumer_mixed)
if(WIN32 AND BUILD_SHARED_LIBS)
  string(REPLACE ";" "\\;" consumer_path "$ENV{PATH}")
  set_tests_properties(consumer-variables consumer-calls consumer-session consumer-mixed PROPERTIES
    ENVIRONMENT "PATH=$<TARGET_FILE_DIR:MIRNext::MIRNext>\\;${consumer_path}")
endif()
